function calcDeriv(){
  const agi=parseInt(document.getElementById('agilidade').value)||0;
  const vig=parseInt(document.getElementById('vigor').value)||0;
  const pre=parseInt(document.getElementById('presenca').value)||0;
  const def=10+agi+(parseInt(document.getElementById('defBonus').value)||0);
  document.getElementById('defTotal').textContent=def;
  const cls=document.getElementById('classe').value||'especialista';
  const nex=parseInt(document.getElementById('nexN').value)||5;
  const s=CLASSES_STATS[cls];
  const niveis=Math.floor(nex/5);
  const pvMax=s.pvBase+vig+(niveis-1)*(s.pvNex+vig);
  const peMax=s.peBase+pre+(niveis-1)*(s.peNex+pre);
  const sanMax=s.sanBase+(niveis-1)*s.sanNex;
  document.getElementById('pvMax').value=pvMax;
  document.getElementById('peMax').value=peMax;
  document.getElementById('sanMax').value=sanMax;
  document.getElementById('pvFormula').textContent=`${s.pvBase}+Vig base (${cls}), +${s.pvNex}+Vig/NEX`;
  document.getElementById('peFormula').textContent=`${s.peBase}+Pre base (${cls}), +${s.peNex}+Pre/NEX`;
  document.getElementById('sanFormula').textContent=`${s.sanBase} base (${cls}), +${s.sanNex}/NEX`;
  const pvA=parseInt(document.getElementById('pvAtual').value)||0;
  if(pvA===0||pvA>pvMax)document.getElementById('pvAtual').value=pvMax;
  const peA=parseInt(document.getElementById('peAtual').value)||0;
  if(peA===0||peA>peMax)document.getElementById('peAtual').value=peMax;
  const sanA=parseInt(document.getElementById('sanAtual').value)||0;
  if(sanA===0||sanA>sanMax)document.getElementById('sanAtual').value=sanMax;
  atuTodas();atuBarras();triggerSalvar();
}

function d20s(n){
  const q=Math.max(n,1);const r=[];
  for(let i=0;i<q;i++)r.push(Math.floor(Math.random()*20)+1);

}

function switchRitTab(btn,tab){
  const item=btn.closest('.rit-item');
  item.querySelectorAll('.rit-tab').forEach(b=>b.classList.remove('active'));
  item.querySelectorAll('.rit-version').forEach(v=>v.classList.remove('active'));
  btn.classList.add('active');
  item.querySelector(`.rit-version[data-tab="${tab}"]`).classList.add('active');
}

function abrirModalRit(){
  let html=`<div class="modal-ov" id="modalRit">
    <div class="modal-box">

  RITUAIS_LIVRO.forEach((r,i)=>{

function inserirRitual(i){
  const r=RITUAIS_LIVRO[i];
  let n_desc=r.desc;let d_pe='';let d_desc='';let v_pe='';let v_desc='';
  const discM=r.desc.match(/Disc\(([^)]+)\):\s*([^.]+)\./);
  if(discM){const m=discM[1].match(/\+(\d+)PE/);if(m)d_pe=m[1];d_desc=discM[2].trim();}
  const verdM=r.desc.match(/Verd\(([^)]+)\):\s*([^.]+)\./);
  if(verdM){const m=verdM[1].match(/\+(\d+)PE/);if(m)v_pe=m[1];v_desc=verdM[2].trim();}
  if(discM||verdM)n_desc=r.desc.replace(/Disc\([^)]+\):[^.]+\./g,'').replace(/Verd\([^)]+\):[^.]+\./g,'').trim();
  document.getElementById('ritCont').appendChild(mkRit({nome:r.nome,elem:r.elem,circ:r.circ,n_exec:r.exec,n_alc:r.alc,n_dur:r.dur,n_pe:r.pe,n_desc:n_desc,d_exec:r.exec,d_alc:r.alc,d_dur:r.dur,d_pe:d_pe,d_desc:d_desc,v_exec:r.exec,v_alc:r.alc,v_dur:r.dur,v_pe:v_pe,v_desc:v_desc}));
  document.getElementById('modalRit')?.remove();triggerSalvar();
}
function filtrarModal(modalId,query,listId){
  const q=query.toLowerCase().trim();
  document.getElementById(listId)?.querySelectorAll('[data-search]').forEach(el=>{el.style.display=(!q||el.dataset.search.includes(q))?'':'none';});
}

function preencher(f){

  ['pvAtual','pvMax','peAtual','peMax','sanAtual','sanMax','defBonus'].forEach(id=>sv(id,f[id]||0));
  sv('patente',f.patente||'recruta');sv('creditos',f.creditos||'baixo');
  const pt=PATENTES[f.patente||'recruta'];if(pt)document.getElementById('patenteDisplay').textContent=pt.label.toUpperCase();
  document.getElementById('avatarUrl').value=f.avatarUrl||'';setAv(f.avatarUrl||'');
  window._ps=f.pericias?JSON.parse(f.pericias):{};
  window._psBon=f.perBonus?JSON.parse(f.perBonus):{};
  window._profExtras=f.profExtras?JSON.parse(f.profExtras):[];
  buildPeri();
  if(f.profNome&&document.getElementById('pn_profissao'))document.getElementById('pn_profissao').value=f.profNome;
  document.getElementById('atkCont').innerHTML='';(f.ataques||[]).forEach(a=>document.getElementById('atkCont').appendChild(mkAtk(a.n,a.ba,a.d,a.bd)));

function salvarFicha(){setSave('saving');try{localStorage.setItem('op_ficha',JSON.stringify(coletarFicha()));setSave('saved');}catch(e){setSave('error');}}

function exportarJSON(){
  const dados=coletarFicha();const nome=dados.nome||'agente';
  const blob=new Blob([JSON.stringify(dados,null,2)],{type:'application/json'});
  const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download=`ficha-${nome.toLowerCase().replace(/\s+/g,'-')}.json`;a.click();
}
function importarJSON(){
  const inp=document.createElement('input');inp.type='file';inp.accept='.json';
  inp.onchange=e=>{const f=e.target.files[0];if(!f)return;const r=new FileReader();r.onload=ev=>{try{preencher(JSON.parse(ev.target.result));salvarFicha();}catch(err){alert('Arquivo inválido ou corrompido.');}};r.readAsText(f);};inp.click();
}
